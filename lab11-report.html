function clamp(num, min, max) {
  if (!Number.isFinite(num)) return min;
  return Math.min(Math.max(num, min), max);
}

function sanitizeCapacity(rawValue) {
  const str = String(rawValue ?? '').trim();
  if (str.length === 0) {
    return { valid: false, value: clamp(manualCapacity, 10, 250),
      error: 'Поле ємності порожнє. Вкажіть значення від 10 до 250 кВт·год.' };
  }
  if (!/^-?\d+(?:[.,]\d+)?$/.test(str)) {
    return { valid: false, value: clamp(manualCapacity, 10, 250),
      error: 'Дозволено лише числа (наприклад, 65). Нечислові символи заблоковано.' };
  }
  const parsed = parseFloat(str.replace(',', '.'));
  if (!Number.isFinite(parsed) || Number.isNaN(parsed)) {
    return { valid: false, value: 65, error: 'Некоректне числове значення.' };
  }
  if (parsed <= 0) {
    return { valid: false, value: 10,
      error: 'Ємність не може бути ≤ 0. Для безпеки застосовано мінімум 10 кВт·год.' };
  }
  if (parsed < 10) {
    return { valid: false, value: 10, error: 'Мінімальна ємність батареї — 10 кВт·год.' };
  }
  if (parsed > 250) {
    return { valid: false, value: 250,
      error: 'Перевищено максимум (250 кВт·год). Обмежено для захисту від переповнення.' };
  }
  return { valid: true, value: Math.round(parsed * 10) / 10, error: '' };
}
